<h2 className="mb-4 text-[20px] font-semibold text-black dark:text-white">Button state</h2>

```tsx
import { useState } from "react";
import { Pressable } from "react-native";
import Laminar from "react-native-laminar";

export function SaveButton() {
  const [saved, setSaved] = useState(false);

  return (
    <Pressable onPress={() => setSaved(true)}>
      <Laminar
        text={saved ? "Saved" : "Save changes"}
        animationPreset="smooth"
      />
    </Pressable>
  );
}
```

<section id="balance" className="mb-16 scroll-mt-20 px-6">

<h2 className="mb-4 text-[20px] font-semibold text-black dark:text-white">Balance</h2>

```tsx
import Laminar from "react-native-laminar";

export function Balance({ amount }: { amount: number }) {
  const value = new Intl.NumberFormat("en-US", {
    style: "currency",
    currency: "USD",
  }).format(amount);

  return (
    <Laminar
      text={value}
      variant="number"
      align="right"
      fontSize={32}
      animationPreset="snappy"
    />
  );
}
```

</section>
